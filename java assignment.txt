2.class BankAccount:
    def __init__(self, accountNumber, accountHolderName, balance=0.0):
        self.accountNumber = accountNumber
        self.accountHolderName = accountHolderName
        self.balance = balance
    def deposit(self, amount):
        if amount > 0:
            self.balance += amount
            print(f"Deposited: ${amount:.2f}. New balance: ${self.balance:.2f}")
        else:
            print("Deposit amount must be positive.")
    def withdraw(self, amount):
        if 0 < amount <= self.balance:
            self.balance -= amount
            print(f"Withdrew: ${amount:.2f}. New balance: ${self.balance:.2f}")
        else:
            print("Withdrawal amount must be positive and cannot exceed the current balance.")
    def display_account_info(self):
        print(f"Account Number: {self.accountNumber}")
        print(f"Account Holder Name: {self.accountHolderName}")
        print(f"Balance: ${self.balance:.2f}")
if __name__ == "__main__":
    account1 = BankAccount("123456", "Alice Smith")
    account1.display_account_info()
    account1.deposit(1000)
    account1.withdraw(500)
    account1.display_account_info()
    account2 = BankAccount("789012", "Bob Johnson", 500.0)
    account2.display_account_info()
3.class Library:
    def __init__(self, libraryName, location=None):
        self.libraryName = libraryName
        self.location = location if location else "Unknown"
        self.books = {}
        self.books_lent = 0
    def add_book(self, title, quantity):
        if title in self.books:
            self.books[title] += quantity
        else:
            self.books[title] = quantity
    def lend_book(self, title):
        if title in self.books and self.books[title] > 0:
            self.books[title] -= 1
            self.books_lent += 1
            print(f"{title} lent successfully.")
        else:
            print(f"{title} is not available for lending.")
    def display_details(self):
        print(f"Library Name: {self.libraryName}")
        print(f"Location: {self.location}")
        print(f"Total Books: {sum(self.books.values())}")
        print(f"Books Lent: {self.books_lent}")
        print(f"Remaining Books: {sum(quantity for quantity in self.books.values())}")
if __name__ == "__main__":
    library = Library("City Library", "Downtown")
    library.add_book("Python Programming", 5)
    library.add_book("Data Structures", 3)
    library.lend_book("Python Programming")
    library.display_details()
4.class Student:
    def __init__(self, name, rollNumber, department=None, semester=None):
        self.name = name
        self.rollNumber = rollNumber
        self.department = department
        self.semester = semester
    def calculate_gpa(self, grades):
        if not grades:
            return 0.0
        return sum(grades) / len(grades)
    def display_info(self):
        print(f"Name: {self.name}")
        print(f"Roll Number: {self.rollNumber}")
        if self.department:
            print(f"Department: {self.department}")
        if self.semester:
            print(f"Semester: {self.semester}")
if __name__ == "__main__":
    student1 = Student("Alice", "S123", "Computer Science", 2)
    student1.display_info()
    gpa = student1.calculate_gpa([3.5, 4.0, 3.8])
    print(f"GPA: {gpa:.2f}")

7.class Display:
    def __init__(self):
        self.a = 0
        self.b = 0
        self.c = 0
    def Accept(self):
        self.a = int(input("Enter the first number (a): "))
        self.b = int(input("Enter the second number (b): "))
    def Max(self):
        self.c = max(self.a, self.b)
        print(f"The greater of {self.a} and {self.b} is: {self.c}")
    def Min(self):
        self.c = min(self.a, self.b)
        print(f"The smaller of {self.a} and {self.b} is: {self.c}")

        self.c = abs(self.a - self.b)

10.class Finder:
    def findMax(self, n1, n2, n3=None):
        if n3 is not None:
            return max(n1, n2, n3)
        else:
            return max(n1, n2)
if __name__ == "__main__":

11.import random
class Game:
    def __init__(self):
        self.players = []
    def accept_player_details(self):
        for i in range(2):
            name = input(f"Enter the name of player {i + 1}: ")
            self.players.append({"name": name, "score": 0, "throws": 0})
    def play_game(self):
        for player in self.players:
            while player["score"] < 20:
                throw = random.randint(1, 6)  # Simulate throwing a dice
                player["score"] += throw
                player["throws"] += 1
                print(f"{player['name']} threw a {throw}. Total score: {player['score']}")
    def display_result(self):
        winner = min(self.players, key=lambda p: (p["score"] < 20, p["throws"]))
        print("\nGame Over!")
        for player in self.players:
            print(f"{player['name']} - Final Score: {player['score']}, Throws: {player['throws']}")
        print(f"Winner: {winner['name']} with {winner['throws']} throws!")
if __name__ == "__main__":
    game = Game()
    game.accept_player_details()
    game.play_game()
    game.display_result()
12.class ParkingLot:
    def __init__(self):
        self.vno = input("Enter vehicle number: ")
        self.hours = int(input("Enter number of hours parked: "))
        self.bill = 0.0
    def calculate(self):
        if self.hours <= 1:
            self.bill = 30  # Fixed charge for the first hour
        else:
            self.bill = 30 + (self.hours - 1) * 1.50  # Charge for additional hours
        return self.bill
    def display_info(self):
        print(f"Vehicle Number: {self.vno}")
        print(f"Hours Parked: {self.hours}")
        print(f"Total Bill: Rs.{self.bill:.2f}")
if __name__ == "__main__":
    parking_lot = ParkingLot()
    parking_lot.calculate()
    parking_lot.display_info()
